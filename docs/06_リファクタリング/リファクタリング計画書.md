// hooks/useJobPolling.ts
export function useJobPolling<TStatus extends { status: string }>(params: {
    source: LibrarySource;
    statusEndpoint: string;
    idleStatus: TStatus;
    onComplete?: () => void;
}): { jobStatus: TStatus; startPolling: () => void; clearPolling: () => void };

// useAutoFillAuthors.ts (~30 行に圧縮)
export function useAutoFillAuthors(source, onComplete) {
    const { jobStatus, startPolling, clearPolling } = useJobPolling<AutoFillStatus>({...});
    const startAutoFill = async (mode) => {
        await apiClient.post(API_ENDPOINTS.META_AUTO_FILL, null, { params: { source, mode } });
        startPolling();
    };
    return { jobStatus, startAutoFill };
}

### Phase 35（2026-05-01）— backend ジョブ状態管理パターンの共通化（P3）

**問題**: `auto_fill_service.py` と `series_resolver.py` で、ソース別ジョブ状態を管理する以下のパターンが完全に重複している。

```python
# auto_fill_service.py
_auto_fill_states: dict[str, AutoFillState] = {}
_auto_fill_states_lock = threading.Lock()

def get_auto_fill_state(source: str) -> AutoFillState:
    with _auto_fill_states_lock:
        if source not in _auto_fill_states:
            _auto_fill_states[source] = AutoFillState()
        return _auto_fill_states[source]

def reset_auto_fill_state(source: str) -> None:
    with _auto_fill_states_lock:
        _auto_fill_states[source] = AutoFillState(status="running")
```

`series_resolver.py` にも同じ構造（`_states` / `_states_lock` / `get_state` / `reset_state`）が存在。

**改善案**: ジェネリッククラス `JobStateManager[T]` を `services/job_state.py` に新設し、両サービスから利用。

```python
# services/job_state.py
class JobStateManager(Generic[T]):
    def __init__(self, factory: Callable[[], T]):
        self._factory = factory
        self._states: dict[str, T] = {}
        self._lock = threading.Lock()

    def get(self, source: str) -> T:
        with self._lock:
            if source not in self._states:
                self._states[source] = self._factory()
            return self._states[source]

    def reset(self, source: str, **initial_kwargs: Any) -> None:
        with self._lock:
            self._states[source] = self._factory(**initial_kwargs)

# auto_fill_service.py
_auto_fill_manager = JobStateManager(AutoFillState)

def get_auto_fill_state(source): return _auto_fill_manager.get(source)
def reset_auto_fill_state(source): _auto_fill_manager.reset(source, status="running")
```

|---|---|---|
| 35-1 | `services/job_state.py` に `JobStateManager` を新設 | P3 |
| 35-2 | `auto_fill_service` / `series_resolver` を移行 | P3 |

**変更対象**: `backend/services/auto_fill_service.py` / `backend/services/series_resolver.py`
**新規作成**: `backend/services/job_state.py`
**完了条件**: 重複した `_states` dict + `_lock` + `get_state` + `reset_state` パターンが 1 箇所に集約。既存テストが通る。

### Phase 38（2026-05-01）— Bulk Dialog 共通サブミットフック（P3）

```tsx
const [saving, setSaving] = useState(false);
const [error, setError] = useState<string | null>(null);

const handleSubmit = async () => {
    setError(null);
    setSaving(true);
    try {
        await onApply(...);
        onClose();
    } catch (e: unknown) {
        setError(e instanceof Error ? e.message : '保存に失敗しました。');
    } finally {
        setSaving(false);
    }
};
```

**改善案**: `useDialogSubmit({ onSubmit, errorMessage, onClose })` フックを新設して saving/error 管理を共通化。

```ts
// hooks/useDialogSubmit.ts
export function useDialogSubmit<T>(params: {
    onSubmit: (value: T) => Promise<void>;
    onClose: () => void;
    errorMessage?: string;
}): {
    saving: boolean;
    error: string | null;
    submit: (value: T) => Promise<void>;
    setError: (msg: string | null) => void;
};
```

各ダイアログは `submit(authors)` のように呼ぶだけ。試行錯誤の余地が大きいので **Phase 38 単独実施を推奨**（他フェーズと混ぜない）。
